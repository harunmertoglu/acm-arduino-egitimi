void cikislariAyarla(byte seviye) {
  for (byte i = 0; i < BUZZER_SAYISI; i++) {
    digitalWrite(PINLER[i], seviye);
  }
}

void notaCal(uint16_t frekans, unsigned long sureMs) {
  if (frekans == SUS) {
    cikislariAyarla(LOW);
    delay(sureMs);
    return;
  }

  const unsigned long yarimPeriyotUs =
      500000UL / frekans;

  const unsigned long boslukMs = 15;
  const unsigned long sesSuresi = sureMs - boslukMs;
  const unsigned long baslangic = millis();

  while (millis() - baslangic < sesSuresi) {
    unsigned long kenar = micros();

    cikislariAyarla(HIGH);
    while (micros() - kenar < yarimPeriyotUs) {}

    kenar = micros();

    cikislariAyarla(LOW);
    while (micros() - kenar < yarimPeriyotUs) {}
  }

  cikislariAyarla(LOW);
  delay(boslukMs);
}

void kalipCal(byte kalipNo) {
  const unsigned long birimMs = 60000UL / BPM / 4;

  for (byte i = 0; i < 16; i++) {
    Nota nota;

    memcpy_P(
      &nota,
      &kaliplar[kalipNo][i],
      sizeof(Nota)
    );

    // Ölçünün sonu
    if (nota.sure == 0) {
      break;
    }

    notaCal(nota.frekans, birimMs * nota.sure);
  }
}

void bolumCal(const byte* bolum, unsigned int uzunluk) {
  for (unsigned int i = 0; i < uzunluk; i++) {
    kalipCal(pgm_read_byte(bolum + i));
  }
}

void setup() {
  for (byte i = 0; i < BUZZER_SAYISI; i++) {
    pinMode(PINLER[i], OUTPUT);
    digitalWrite(PINLER[i], LOW);
  }
}

void loop() {
  bolumCal(ilkBolum, sizeof(ilkBolum));
  bolumCal(basaDonus, sizeof(basaDonus));
  bolumCal(coda, sizeof(coda));

  delay(3000);
}
