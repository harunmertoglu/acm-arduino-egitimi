/*
Bir Derdim Var — Mor ve Ötesi (tam bölüm)
ARDUINO UNO — 1 PASİF BUZZER

BAĞLANTI
  Buzzer 1: + uç D7, - uç GND
  Çıkış pinlerini birbirine bağlamayın.
  Bağlantıları kartın enerjisini keserek değiştirin.

YÜKLEME
  1. Arduino IDE'de yeni bir proje açın.
  2. Bu dosyanın tamamını kopyalayıp editöre yapıştırın.
  3. Arduino Uno kartını ve kartınızın portunu seçin.
  4. Yükle düğmesine basın. Şarkı otomatik başlar ve tekrar eder.
  5. BPM değerini artırmak hızlandırır, azaltmak yavaşlatır.

NOTLAR
  Pasif buzzer gerekir. Doğrudan bağlantı uygun düşük akımlı piezo içindir;
  daha fazla akım gerektiren parçalar için sürücü kullanın.
  Ritimler yaklaşık aktarılmıştır. Donanımda dinleme testi yapılmamıştır.
  Tüm bağlı buzzerlar aynı melodiyi birlikte çalar.
*/

void tumBuzzerlariAyarla(byte seviye) {
  for (byte i = 0; i < buzzerSayisi; i++) {
    digitalWrite(buzzerPinleri[i], seviye);
  }
}

void ikiBuzzerCal(int frekans, unsigned long sureMs) {
  if (frekans == 0) {
    tumBuzzerlariAyarla(LOW);
    delay(sureMs);
    return;
  }
  const unsigned long yarimPeriyotUs = 500000UL / frekans;
  const unsigned long sesSuresi = sureMs * 9 / 10;
  const unsigned long baslangic = millis();
  while (millis() - baslangic < sesSuresi) {
    unsigned long kenar = micros();
    tumBuzzerlariAyarla(HIGH);
    while (micros() - kenar < yarimPeriyotUs) {}
    kenar = micros();
    tumBuzzerlariAyarla(LOW);
    while (micros() - kenar < yarimPeriyotUs) {}
  }
  tumBuzzerlariAyarla(LOW);
  delay(sureMs - sesSuresi);
}

void kalipCal(byte kalipNo) {
  const unsigned long birimMs = 60000UL / BPM / 4;

  for (byte i = 0; i < 8; i++) {
    Nota nota;
    memcpy_P(&nota, &kaliplar[kalipNo][i], sizeof(Nota));

    if (nota.sure == 0) {
      break;
    }

    ikiBuzzerCal(nota.frekans, birimMs * nota.sure);
  }
}

void setup() {
  for (byte i = 0; i < buzzerSayisi; i++) {
    pinMode(buzzerPinleri[i], OUTPUT);
    digitalWrite(buzzerPinleri[i], LOW);
  }
}

void loop() {
  // Görselin açıklaması: ilk iki ölçüyü 7 kez çal.
  for (byte tekrar = 0; tekrar < 7; tekrar++) {
    kalipCal(0);
    kalipCal(0);
  }

  // 3. ve 4. ölçüler
  kalipCal(0);
  kalipCal(1);

  // Şarkının kalan kısmı
  for (unsigned int i = 0; i < sizeof(sira); i++) {
    kalipCal(pgm_read_byte(&sira[i]));
  }

  delay(3000);
}
