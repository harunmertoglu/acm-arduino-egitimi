void cikislariAyarla(byte seviye) {
  for (byte i = 0; i < BUZZER_SAYISI; i++) {
    digitalWrite(PINLER[i], seviye);
  }
}

void notaCal(uint16_t frekans, unsigned long sureMs) {
  if (frekans == SUS) {
    cikislariAyarla(LOW);
    delay(sureMs);
    return;
  }

  const unsigned long yarimPeriyotUs =
      500000UL / frekans;

  const unsigned long boslukMs = 15;
  const unsigned long sesSuresi = sureMs - boslukMs;
  const unsigned long baslangic = millis();

  while (millis() - baslangic < sesSuresi) {
    unsigned long kenar = micros();

    cikislariAyarla(HIGH);
    while (micros() - kenar < yarimPeriyotUs) {}

    kenar = micros();

    cikislariAyarla(LOW);
    while (micros() - kenar < yarimPeriyotUs) {}
  }

  cikislariAyarla(LOW);
  delay(boslukMs);
}

// Ölçü numaraları 1'den başlar.
void olcuCal(byte olcuNo) {
  const unsigned long birimMs = 60000UL / BPM / 4;

  for (byte i = 0; i < 16; i++) {
    Nota nota;

    memcpy_P(
      &nota,
      &olculer[olcuNo - 1][i],
      sizeof(Nota)
    );

    if (nota.sure == 0) {
      break;
    }

    notaCal(nota.frekans, birimMs * nota.sure);
  }
}

void bolumCal(byte ilk, byte son) {
  for (byte olcu = ilk; olcu <= son; olcu++) {
    olcuCal(olcu);
  }
}

void setup() {
  for (byte i = 0; i < BUZZER_SAYISI; i++) {
    pinMode(PINLER[i], OUTPUT);
    digitalWrite(PINLER[i], LOW);
  }
}

void loop() {
  // Giriş ve ilk tekrarın birinci bitişi
  bolumCal(1, 8);

  // Tekrar: 8. ölçüyü atlayıp ikinci bitişe geç
  bolumCal(5, 7);
  bolumCal(9, 25);

  // Nakarat ve birinci bitiş
  bolumCal(26, 29);

  // Nakarat tekrarı ve ikinci bitiş
  bolumCal(26, 27);
  bolumCal(30, 32);

  delay(3000);
}
